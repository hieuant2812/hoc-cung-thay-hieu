/**
 * MÁY CHỦ NHỎ "LỚP HỌC" — Học cùng Thầy Hiếu
 * Dán toàn bộ mã này vào Google Apps Script của một Google Trang tính (Tiện ích mở rộng → Apps Script).
 * Bước 1: đổi KHOA_GV bên dưới thành mật khẩu riêng của thầy (chỉ thầy biết, dùng trên trang lophoc.html).
 * Bước 2: Triển khai → Tùy chọn triển khai mới → Loại: Ứng dụng web
 *         Thực thi với tư cách: Tôi · Người có quyền truy cập: Bất kỳ ai → Triển khai → chép "URL ứng dụng web".
 * Bước 3: dán URL đó vào config.js (SITE.lopHoc.url) và vào trang lophoc.html.
 * Mỗi khi sửa mã: Triển khai → Quản lý các lần triển khai → Chỉnh sửa (bút chì) → Phiên bản: Mới → Triển khai.
 */
const KHOA_GV = "doi-mat-khau-nay";

const BANG = {
  KetQua: ["Thời gian", "Lớp", "Họ tên", "Loại", "Nội dung", "Điểm", "Số câu", "Chi tiết", "Mã bài", "Trạng thái", "Mã"],
  GiaoBai: ["Mã", "Lớp", "Tiêu đề", "Đường link", "Hạn", "Ghi chú", "Ngày giao"]
};

function bang_(ten) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(ten);
  if (!sh) {
    sh = ss.insertSheet(ten);
    sh.appendRow(BANG[ten]);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, BANG[ten].length).setFontWeight("bold").setBackground("#e8faf6");
  }
  return sh;
}

function json_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    let d = JSON.parse(e.postData.contents || "[]");
    if (!Array.isArray(d)) d = [d];
    const kq = bang_("KetQua");
    // chống ghi trùng khi học sinh gửi lại
    const n = kq.getLastRow();
    const daCo = new Set(n > 1 ? kq.getRange(Math.max(2, n - 1500), 11, Math.min(1500, n - 1), 1).getValues().map(r => String(r[0])) : []);
    const dong = [];
    d.forEach(x => {
      if (!x || typeof x !== "object") return;
      if (x.loai === "giaobai" || x.loai === "xoagiao") {
        if (x.khoa !== KHOA_GV) return;
        const gb = bang_("GiaoBai");
        if (x.loai === "giaobai") gb.appendRow([x.ma || Utilities.getUuid().slice(0, 8), String(x.lop || "TẤT CẢ").toUpperCase(), x.tieude || "", x.link || "", x.han || "", x.ghichu || "", new Date()]);
        else {
          const v = gb.getDataRange().getValues();
          for (let i = v.length - 1; i >= 1; i--) if (String(v[i][0]) === String(x.ma)) gb.deleteRow(i + 1);
        }
        return;
      }
      if (daCo.has(String(x.ma))) return;
      daCo.add(String(x.ma));
      dong.push([x.tg ? new Date(x.tg) : new Date(), String(x.lop || "").toUpperCase(), String(x.hoten || "").slice(0, 60), x.loai || "", String(x.noidung || "").slice(0, 200),
        x.diem == null ? "" : x.diem, x.socau == null ? "" : x.socau, x.chitiet ? JSON.stringify(x.chitiet).slice(0, 4000) : "", x.ma_bai || "", x.trangthai || "", String(x.ma || "")]);
    });
    if (dong.length) kq.getRange(kq.getLastRow() + 1, 1, dong.length, dong[0].length).setValues(dong);
    return json_({ ok: true, them: dong.length });
  } catch (err) {
    return json_({ ok: false, loi: String(err) });
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  const p = e.parameter || {};
  if (p.a === "giaobai") {
    const v = bang_("GiaoBai").getDataRange().getValues().slice(1);
    const lop = String(p.lop || "").toUpperCase();
    const homNay = new Date(); homNay.setHours(0, 0, 0, 0);
    const ds = v.filter(r => p.khoa === KHOA_GV || ((r[1] === lop || r[1] === "TẤT CẢ" || (r[1] && lop.indexOf(r[1]) === 0 && /^\d+$/.test(r[1]))) && (!r[4] || new Date(r[4]) >= homNay)))
      .map(r => ({ ma: String(r[0]), lop: r[1], tieude: r[2], link: r[3], han: r[4] ? Utilities.formatDate(new Date(r[4]), "Asia/Ho_Chi_Minh", "dd/MM/yyyy") : "", ghichu: r[5], ngay: r[6] }));
    return json_({ ok: true, ds: ds.reverse() });
  }
  if (p.a === "ketqua") {
    if (p.khoa !== KHOA_GV) return json_({ ok: false, loi: "Sai khoá giáo viên" });
    const v = bang_("KetQua").getDataRange().getValues().slice(1);
    const tu = p.tu ? new Date(p.tu) : null;
    const ds = v.filter(r => !tu || new Date(r[0]) >= tu).slice(-20000).map(r => ({ tg: r[0], lop: r[1], hoten: r[2], loai: r[3], noidung: r[4], diem: r[5], socau: r[6], chitiet: r[7], ma_bai: r[8], trangthai: r[9] }));
    return json_({ ok: true, ds: ds });
  }
  return json_({ ok: true, ten: "Lớp học – Học cùng Thầy Hiếu" });
}
